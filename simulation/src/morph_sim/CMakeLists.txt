cmake_minimum_required(VERSION 3.16)
project(morph_sim)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ros_gz_bridge REQUIRED)
add_executable(lifecycle_bridge src/lifecycle_bridge.cpp)
target_compile_features(lifecycle_bridge PUBLIC cxx_std_17)
target_link_libraries(lifecycle_bridge ros_gz_bridge::ros_gz_bridge rclcpp::rclcpp)
install(TARGETS lifecycle_bridge DESTINATION lib/${PROJECT_NAME})
if(NOT "$ENV{ROS_DISTRO}" STREQUAL "humble")
 find_package(gz_sim_vendor REQUIRED)
 find_package(gz_transport_vendor REQUIRED)
 find_package(gz_msgs_vendor REQUIRED)
 find_package(gz_plugin_vendor REQUIRED)
 find_package(gz_gui_vendor REQUIRED)
endif()
if("$ENV{ROS_DISTRO}" STREQUAL "lyrical")
 find_package(gz-sim REQUIRED)
 find_package(gz-transport REQUIRED)
 find_package(gz-msgs REQUIRED)
 find_package(gz-plugin-register REQUIRED)
 find_package(gz-gui REQUIRED)
 find_package(Qt6 COMPONENTS Core Quick REQUIRED)
 set(MORPH_QT Qt6::Core Qt6::Quick)
 set(MORPH_GUI gz-gui::gz-gui)
 set(MORPH_GZ_LIBS gz-sim::gz-sim gz-transport::gz-transport gz-msgs::gz-msgs gz-plugin::gz-plugin-register)
else()
 find_package(gz-sim8 REQUIRED)
 find_package(gz-transport13 REQUIRED)
 find_package(gz-msgs10 REQUIRED)
 find_package(gz-plugin2 REQUIRED COMPONENTS register)
 find_package(gz-gui8 REQUIRED)
 find_package(Qt5 COMPONENTS Core Quick REQUIRED)
 set(MORPH_QT Qt5::Core Qt5::Quick)
 set(MORPH_GUI gz-gui8::gz-gui8)
 set(MORPH_GZ_LIBS gz-sim8::gz-sim8 gz-transport13::gz-transport13 gz-msgs10::gz-msgs10 gz-plugin2::register)
endif()
add_library(FrameMeter SHARED src/FrameMeter.cpp src/frame_meter.qrc)
set_target_properties(FrameMeter PROPERTIES AUTOMOC ON AUTORCC ON)
target_compile_features(FrameMeter PUBLIC cxx_std_17)
target_link_libraries(FrameMeter ${MORPH_GUI} ${MORPH_QT} ${MORPH_GZ_LIBS})
install(TARGETS FrameMeter LIBRARY DESTINATION lib)
add_library(morph_suspension SHARED src/suspension_system.cpp)
target_compile_features(morph_suspension PUBLIC cxx_std_17)
target_link_libraries(morph_suspension ${MORPH_GZ_LIBS})
add_library(morph_watchdog SHARED src/drive_watchdog.cpp)
target_compile_features(morph_watchdog PUBLIC cxx_std_17)
target_link_libraries(morph_watchdog ${MORPH_GZ_LIBS})
add_library(morph_height_servo SHARED src/height_servo.cpp)
target_compile_features(morph_height_servo PUBLIC cxx_std_17)
target_link_libraries(morph_height_servo ${MORPH_GZ_LIBS})
install(TARGETS morph_suspension morph_watchdog morph_height_servo LIBRARY DESTINATION lib)
ament_python_install_package(morph_sim)
install(PROGRAMS nodes/perception_adapter nodes/rviz_state nodes/control_console nodes/camera_environment nodes/interaction_runtime nodes/terrain_manager nodes/teleop nodes/gazebo_keyboard nodes/camera_gate nodes/demo_sequence nodes/diagnose_motion nodes/joint_control nodes/smoke_test DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY urdf worlds config launch meshes models DESTINATION share/${PROJECT_NAME})
install(DIRECTORY src DESTINATION share/${PROJECT_NAME})
install(DIRECTORY tools DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(ament_cmake_pytest REQUIRED)
 ament_add_pytest_test(perception tests/test_perception.py)
 ament_add_pytest_test(core tests/test_core.py)
 ament_add_pytest_test(physics tests/test_physics.py)
 ament_add_pytest_test(parameters tests/test_parameters.py)
 ament_add_pytest_test(safety tests/test_safety.py)
 ament_add_pytest_test(experiment_batch tests/test_experiment_batch.py)
 ament_add_pytest_test(sequence_gaps tests/test_sequence_gaps.py)
endif()
install(FILES README.md VALIDATION_STATUS.json LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
